if(NOT EXISTS "${PROJECT_SOURCE_DIR}/3rdparty/SPIRV-Cross/spirv_msl.hpp")
    message(FATAL_ERROR "Missing SPIRV-Cross. Run git submodule update --init --recursive.")

set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_GLSL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_MSL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${PROJECT_SOURCE_DIR}/3rdparty/SPIRV-Cross" "${CMAKE_CURRENT_BINARY_DIR}/spirv-cross" EXCLUDE_FROM_ALL)

target_link_libraries(anyps5_metal PUBLIC shader_recompiler_host PRIVATE spirv-cross-msl "-framework Metal" "-framework Foundation")
target_compile_options(anyps5_metal PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
set_target_properties(anyps5_metal PROPERTIES CXX_EXTENSIONS OFF OBJCXX_EXTENSIONS OFF)

add_executable(agc_metal_shader MetalShaderTool.cpp)
target_link_libraries(agc_metal_shader PRIVATE anyps5_metal)
